Nodo* construir(vector<char>& d, int i) {


  if (i >= d.size()) return nullptr;


  Nodo* n = new Nodo(d[i]);


  n->izq = construir(d, 2 * i + 1);


  n->der = construir(d, 2 * i + 2);


  return n;

}
