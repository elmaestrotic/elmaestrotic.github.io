#include

3. Insertar un nodo al final

📝 Enunciado: Escribe una función insertarFinal que recorra la lista hasta el último nodo y añada un nuevo nodo al final, como se muestra en la presentación.

#include 
struct Nodo {
    int dato;
    Nodo* siguiente;
};
void insertarFinal(Nodo*& primero, int valor) {
    Nodo* nuevo = new Nodo;
    nuevo->dato = valor;
    nuevo->siguiente = NULL;
    if (primero == NULL) {
        primero = nuevo;
    } else {
        Nodo* aux = primero;
        while (aux->siguiente != NULL) {
            aux = aux->siguiente;
        }
        aux->siguiente = nuevo;
    }
}
int main() {
    Nodo* primero = NULL;
    insertarFinal(primero, 10);
    std::cout << "Nodo insertado al final, valor = " << primero->dato << std::endl;
    return 0;
}

4. Contar nodos en la lista

📝 Enunciado: Crea una función contarNodos que recorra la lista y devuelva el número de nodos, similar a la gestión de tamaño mencionada en la presentación.

#include 
struct Nodo {
    int dato;
    Nodo* siguiente;
};
int contarNodos(Nodo* primero) {
    int contador = 0;
    Nodo* aux = primero;
    while (aux != NULL) {
        contador++;
        aux = aux->siguiente;

    return contador;
}

    return 0;

5. Buscar un elemento

📝 Enunciado: Implementa una función buscarElemento que recorra la lista y devuelva true si encuentra un valor dado, false en caso contrario.

#include 
struct Nodo {
    int dato;
    Nodo* siguiente;
};
bool buscarElemento(Nodo* primero, int valor) {
    Nodo* aux = primero;
    while (aux != NULL) {
        if (aux->dato == valor) return true;
        aux = aux->siguiente;
    }
    return false;
}
int main() {
    Nodo* primero = NULL;
    std::cout << "Elemento 5 encontrado: " << (buscarElemento(primero, 5) ? "Sí" : "No") << std::endl;
    return 0;
}

6. Eliminar el primer nodo

📝 Enunciado: Escribe una función eliminarPrimero que elimine el primer nodo de la lista y libere la memoria, actualizando el puntero primero.

#include 
struct Nodo {
    int dato;
    Nodo* siguiente;
};
void eliminarPrimero(Nodo*& primero) {
    if (primero == NULL) return;

    primero = primero->siguiente;

}

7. Insertar en posición específica

📝 Enunciado: Crea una función insertarPosicion que inserte un nodo con un valor dado en una posición específica, recorriendo la lista con un puntero auxiliar.

#include 
struct Nodo {
    int dato;
    Nodo* siguiente;
};
void insertarPosicion(Nodo*& primero, int valor, int posicion) {
    if (posicion < 0) return;
    Nodo* nuevo = new Nodo{valor, NULL};
    if (posicion == 0) {
        nuevo->siguiente = primero;
        primero = nuevo;
    } else {
        Nodo* aux = primero;
        for (int i = 0; i < posicion - 1 && aux != NULL; i++) {
            aux = aux->siguiente;
        }
        if (aux != NULL) {
            nuevo->siguiente = aux->siguiente;
            aux->siguiente = nuevo;
        } else {
            delete nuevo;
        }
    }
}
int main() {
    Nodo* primero = NULL;
    insertarPosicion(primero, 10, 0);
    std::cout << "Nodo insertado en posición 0, valor = " << primero->dato << std::endl;
    return 0;
}

8. Mostrar lista

    Nodo* siguiente;

9. Invertir la lista

📝 Enunciado: Escribe una función invertirLista que invierta los enlaces de la lista simplemente enlazada, actualizando el puntero primero.

#include 
struct Nodo {
    int dato;
    Nodo* siguiente;
};
void invertirLista(Nodo*& primero) {
    Nodo* prev = NULL;
    Nodo* actual = primero;
    Nodo* siguiente = NULL;
    while (actual != NULL) {
        siguiente = actual->siguiente;
        actual->siguiente = prev;
        prev = actual;
        actual = siguiente;
    }
    primero = prev;
}
int main() {
    Nodo* primero = new Nodo{1, new Nodo{2, NULL}};
    invertirLista(primero);
    std::cout << "Lista invertida, primer valor = " << primero->dato << std::endl;
    return 0;
}

10. Liberar memoria de la lista

📝 Enunciado: Crea una función liberarLista que recorra la lista, libere la memoria de cada nodo y deje el puntero primero en NULL.

#include 
struct Nodo {
    int dato;
    Nodo* siguiente;
};
void liberarLista(Nodo*& primero) {
    Nodo* actual = primero;
    while (actual != NULL) {
        Nodo* temp = actual;
        actual = actual->siguiente;
        delete temp;
    }
    primero = NULL;
}
int main() {
    Nodo* primero = new Nodo{1, new Nodo{2, NULL}};
    liberarLista(primero);
    std::cout << "Lista liberada, primero = " << primero << std::endl;
    return 0;
}
